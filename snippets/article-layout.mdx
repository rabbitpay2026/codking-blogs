export const Share = () => {
  const dot = "flex h-8 w-8 items-center justify-center rounded-full text-xs no-underline";
  const share = (build) => (e) => {
    if (typeof window === "undefined") return;
    e.preventDefault();
    window.open(build(window.location.href), "_blank", "noopener");
  };
  const copy = (e) => {
    if (typeof navigator === "undefined" || !navigator.clipboard) return;
    e.preventDefault();
    navigator.clipboard.writeText(window.location.href);
  };
  return (
    <div className="mt-3 flex gap-2">
      <a href="https://www.linkedin.com/sharing/share-offsite/" onClick={share((u) => "https://www.linkedin.com/sharing/share-offsite/?url=" + encodeURIComponent(u))} className={dot} style={{ background: CK.surface, color: CK.ink }} aria-label="Share on LinkedIn">in</a>
      <a href="https://x.com/intent/post" onClick={share((u) => "https://x.com/intent/post?url=" + encodeURIComponent(u))} className={dot} style={{ background: CK.surface, color: CK.ink }} aria-label="Share on X">X</a>
      <a href="/blog" onClick={copy} className={dot} style={{ background: CK.surface, color: CK.ink }} aria-label="Copy link to this article">&#8599;</a>
    </div>
  );
};
